				type = alliance
				id = ROOT
				value = 100
			}
			add_ai_strategy = {
				type = befriend
				id = ROOT
				value = 100
			}
			country_event = { id = generic.2 hours = 18 }
		}
		add_ai_strategy = {
			type = alliance
			id = ENG
			value = 100
		}
		add_ai_strategy = {
			type = befriend
			id = ENG
			value = 100
		}
		ai_chance = {
			base = 75
		}
	}
	option = { #We must preserve our national interests...
		name = nepal.1.b

			limit = {
				is_subject = no
				is_in_faction = no

		}

			}

			custom_effect_tooltip = generic_skip_one_line_tt
			FROM = { country_event = nepal.10 }
			set_country_flag = NEP_accepted_himalayan_defense_pact

				add_political_power = -10
				add_ideas = NEP_himalayan_defense_pact_idea_r56

				custom_effect_tooltip = generic_skip_one_line_tt
				FROM = { country_event = nepal.10 }
				set_country_flag = NEP_accepted_himalayan_defense_pact
			}
		}

		ai_chance = {
			base = 1
			modifier = { # Same Government
				add = 2
				has_government = FROM
			}

			modifier = { # High Opinion
				add = 2
				has_opinion = {
					target = FROM
					value > 50
				}
			}

			modifier = { # Someone is justifying against us
				add = 3
				any_other_country = {
					NOT = { has_country_flag = NEP_accepted_himalayan_defense_pact original_tag = NEP }

				}
			}
		}
	}

		add_political_power = 10

			add_opinion_modifier = {
				target = ROOT

		set_country_flag = NEP_we_dont_want_alliance_with_you_nepalese

			base = 1
			modifier = { # Low Opinion

					target = FROM
					value < 0
				}
			}

			modifier = { # Generated World Tension
				add = 3
				FROM = { has_added_tension_amount > 0 }
			}
		}
	}
}
