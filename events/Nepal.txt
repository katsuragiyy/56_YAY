country_event = {
	id = nepal.1
	title = nepal.1.t
	desc = nepal.1.d
	picture = GFX_report_event_polish_politics

	trigger = {
		original_tag = NEP
		is_subject = no
		is_in_faction = no
		has_capitulated = no
		has_completed_focus = NEP_maintain_british_ties_r56
		OR = {
			has_government = neutrality
			has_government = democratic
		}
		ENG = {
			any_country = {
				OR = {
					tag = GER
					tag = ITA
					tag = JAP
					tag = USA
					tag = SOV
					tag = CHI
					tag = FRA
				}
				has_offensive_war_with = ENG
			}
			OR = {
				has_government = democratic
				has_government = neutrality
			}
			is_faction_leader = yes
			has_capitulated = no
			NOT = {
				has_war_with = NEP
				is_in_faction_with = NEP
			}
		}
		NEP = {
			any_neighbor_country = {
				OR = {
					tag = ENG
					is_subject_of = ENG
					is_in_faction_with = ENG
				}
			}
		}
	}

	mean_time_to_happen = {
		days = 20
	}

	fire_only_once = yes

	option = { #We must aid our allies. Immediately declare war!
		name = nepal.1.a
		ENG = {
			add_opinion_modifier = {
				target = NEP
				modifier = large_increase
			}
			add_ai_strategy = {
				type = alliance
				id = ROOT
				value = 100
			}
			add_ai_strategy = {
				type = befriend
				id = ROOT
				value = 100
			}
			country_event = { id = generic.2 hours = 18 }
		}
		add_ai_strategy = {
			type = alliance
			id = ENG
			value = 100
		}
		add_ai_strategy = {
			type = befriend
			id = ENG
			value = 100
		}
		ai_chance = {
			base = 75
		}
	}
	option = { #We must preserve our national interests...
		name = nepal.1.b
		if = {
			limit = {
				is_subject = no
				is_in_faction = no
				has_war = no
				has_government = neutrality
			}
			add_ideas = neutrality_idea
		}
		ai_chance = {
			base = 25
			modifier = {
				is_historical_focus_on = yes
				factor = 0
			}
		}
	}	
}

country_event = { # Nepal suggests defensive pact
	id = nepal.8
	title = nepal.8.t
	desc = nepal.8.desc
	picture = GFX_report_event_polish_politics
	is_triggered_only = yes

	option = { # Sure
		name = nepal.8.a
		if = {
			limit = { is_faction_leader = yes }

			add_political_power = -10
			FROM = { add_to_faction = ROOT }
			custom_effect_tooltip = generic_skip_one_line_tt
			FROM = { country_event = nepal.10 }
			set_country_flag = NEP_accepted_himalayan_defense_pact
			
			else = {
				add_political_power = -10
				give_guarantee = FROM
				FROM = { give_guarantee = ROOT }

				custom_effect_tooltip = generic_skip_one_line_tt
				FROM = { country_event = nepal.10 }
				set_country_flag = NEP_accepted_himalayan_defense_pact
			}
		}

		ai_chance = {
			base = 1
			modifier = { # Same Government
				add = 2
				has_government = FROM
			}

			modifier = { # High Opinion
				add = 2
				has_opinion = {
					target = FROM
					value > 50
				}
			}

			modifier = { # Someone is justifying against us
				add = 3
				every_other_country = {
					limit = { NOT = { OR = { has_country_flag = NEP_accepted_himalayan_defense_pact original_tag = NEP } } }
					is_justifying_wargoal_against = ROOT
				}
			}
		}
	}

	option = { # no
		name = nepal.8.b
		add_political_power = 10
		FROM = {
			add_opinion_modifier = {
				target = ROOT
				modifier = medium_decrease
			}
		}
		custom_effect_tooltip = generic_skip_one_line_tt
		FROM = { country_event = nepal.9 }

		ai_chance = {
			base = 1

			modifier = { # Low Opinion
				add = 2
				has_opinion = {
					target = FROM
					value < 0
				}
			}

			modifier = { # Generated World Tension
				add = 3
				FROM = { has_added_tension_amount > 0 }
			}
		}
	}
}
